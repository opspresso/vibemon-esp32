cmake_minimum_required(VERSION 3.22)
project(vibemon_tests LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(ARDUINO_LIB_DIR "${CMAKE_SOURCE_DIR}/build/arduino/libraries" CACHE PATH "Isolated Arduino libraries")
if(NOT EXISTS "${ARDUINO_LIB_DIR}/ArduinoJson/src/ArduinoJson.h")
  message(FATAL_ERROR "Install the pinned libraries before configuring native tests")
endif()
add_library(vibemon_core src/domain/model.cpp src/adapters/catalog_json.cpp src/application/controller.cpp)
target_include_directories(vibemon_core PUBLIC src "${ARDUINO_LIB_DIR}/ArduinoJson/src")
target_compile_options(vibemon_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
option(VIBEMON_SANITIZE "Enable native address and undefined-behavior sanitizers" ON)
if(VIBEMON_SANITIZE AND CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(vibemon_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(vibemon_core PUBLIC -fsanitize=address,undefined)
endif()
enable_testing()
add_executable(domain_test tests/domain_test.cpp)
target_link_libraries(domain_test PRIVATE vibemon_core)
target_compile_options(domain_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_test(NAME domain COMMAND domain_test)
add_executable(controller_test tests/controller_test.cpp)
target_link_libraries(controller_test PRIVATE vibemon_core)
target_compile_options(controller_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_test(NAME controller COMMAND controller_test)
