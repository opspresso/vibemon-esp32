cmake_minimum_required(VERSION 3.22)
project(vibemon_tests LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(ARDUINO_LIB_DIR "${CMAKE_SOURCE_DIR}/build/arduino/libraries" CACHE PATH "Isolated Arduino libraries")
if(NOT EXISTS "${ARDUINO_LIB_DIR}/ArduinoJson/src/ArduinoJson.h")
  message(FATAL_ERROR "Install the pinned libraries before configuring native tests")
endif()
add_library(vibemon_core src/domain/model.cpp src/domain/firmware.cpp src/adapters/release_json.cpp src/adapters/catalog_json.cpp src/adapters/cloud_http.cpp src/adapters/settings_json.cpp src/application/controller.cpp src/application/sign_in.cpp src/application/updater.cpp src/adapters/device_auth.cpp)
target_include_directories(vibemon_core PUBLIC src "${CMAKE_SOURCE_DIR}/build/generated" "${ARDUINO_LIB_DIR}/ArduinoJson/src")
target_compile_options(vibemon_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
option(VIBEMON_SANITIZE "Enable native address and undefined-behavior sanitizers" ON)
if(VIBEMON_SANITIZE AND CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(vibemon_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(vibemon_core PUBLIC -fsanitize=address,undefined)
endif()
enable_testing()
add_executable(domain_test tests/domain_test.cpp)
target_link_libraries(domain_test PRIVATE vibemon_core)
target_compile_options(domain_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_test(NAME domain COMMAND domain_test)
add_executable(contract_test tests/contract_test.cpp)
target_link_libraries(contract_test PRIVATE vibemon_core)
target_compile_options(contract_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_test(NAME contract COMMAND contract_test "${CMAKE_SOURCE_DIR}/contracts/cases.json")
add_executable(controller_test tests/controller_test.cpp)
target_link_libraries(controller_test PRIVATE vibemon_core)
target_compile_options(controller_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_test(NAME controller COMMAND controller_test)
add_executable(sign_in_test tests/sign_in_test.cpp)
target_link_libraries(sign_in_test PRIVATE vibemon_core)
target_compile_options(sign_in_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_test(NAME sign_in COMMAND sign_in_test)
add_executable(settings_test tests/settings_test.cpp)
target_link_libraries(settings_test PRIVATE vibemon_core)
target_compile_options(settings_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_test(NAME settings COMMAND settings_test)
add_executable(bounded_body_test tests/bounded_body_test.cpp)
target_link_libraries(bounded_body_test PRIVATE vibemon_core)
target_compile_options(bounded_body_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_test(NAME bounded_body COMMAND bounded_body_test)
add_executable(catalog_check tests/catalog_check.cpp)
target_link_libraries(catalog_check PRIVATE vibemon_core)
target_compile_options(catalog_check PRIVATE -Wall -Wextra -Wpedantic -Werror)

set(LV_BUILD_CONF_PATH "${CMAKE_SOURCE_DIR}/lv_conf.h" CACHE PATH "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
set(LV_USE_FIND_PACKAGE OFF CACHE BOOL "" FORCE)
set(LV_USE_PKG_CONFIG OFF CACHE BOOL "" FORCE)
set(LV_FETCH_DEPENDENCIES OFF CACHE BOOL "" FORCE)
add_subdirectory("${ARDUINO_LIB_DIR}/lvgl" "${CMAKE_BINARY_DIR}/lvgl")
add_library(vibemon_ui src/presentation/assets.cpp src/presentation/dashboard_ui.cpp)
target_include_directories(vibemon_ui PRIVATE "${CMAKE_SOURCE_DIR}/build/generated")
target_link_libraries(vibemon_ui PUBLIC vibemon_core lvgl::lvgl)
target_compile_options(vibemon_ui PRIVATE -Wall -Wextra -Wpedantic -Werror)
if(VIBEMON_SANITIZE AND CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(lvgl PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
endif()
add_executable(ui_test tests/ui_test.cpp)
target_link_libraries(ui_test PRIVATE vibemon_ui)
target_compile_options(ui_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_test(NAME ui COMMAND ui_test)
